{{- /* 鸟种数据注册表与构建期校验（唯一实现）： data/birds.toml 的 id 必须唯一且字段完整；photo.toml 中带 species 的照片 必须能反查到条目；bird_series 必须匹配真实系列；保育等级必须有 i18n 文案。 页面无关，调用方用 partialCached（variant "birds"）只跑一次；返回值供 灯箱壳（labels）与照片卡片（series）复用。 */ -}} {{- $birds := slice -}} {{- $birdSeries := "" -}} {{- with hugo.Data.birds -}} {{- $birdSeries = .bird_series | default "" -}} {{- with .bird }}{{ $birds = . }}{{ end -}} {{- end -}} {{- if not $birdSeries -}} {{- warnf "%s" (i18n "error_bird_series_missing") -}} {{- end -}} {{- $ids := slice -}} {{- $required := slice "name" "latin" "order" "order_latin" "family" "family_latin" "genus" "genus_latin" "distribution" "habitat" "habits" "conservation_level" -}} {{- $levels := slice "lc" "nt" "vu" "en" "cr" "domestic" -}} {{- $labels := dict -}} {{- range $level := $levels -}} {{- $label := i18n (printf "bird_conservation_%s" $level) -}} {{- if not $label -}} {{- errorf "%s" (i18n "error_bird_conservation_i18n_missing" (dict "Level" (printf "%q" $level))) -}} {{- end -}} {{- $labels = merge $labels (dict $level $label) -}} {{- end -}} {{- range $bird := $birds -}} {{- $id := $bird.id | default "" -}} {{- if not $id }}{{ errorf "%s" (i18n "error_bird_id_missing") }}{{ end -}} {{- if in $ids $id }}{{ errorf "%s" (i18n "error_bird_id_duplicate" (dict "ID" (printf "%q" $id))) }}{{ end -}} {{- range $field := $required -}} {{- if not (index $bird $field) }}{{ errorf "%s" (i18n "error_bird_field_missing" (dict "ID" (printf "%q" $id) "Field" (printf "%q" $field))) }}{{ end -}} {{- end -}} {{- if not (in $levels $bird.conservation_level) }}{{ errorf "%s" (i18n "error_bird_conservation_invalid" (dict "ID" (printf "%q" $id) "Level" (printf "%q" $bird.conservation_level))) }}{{ end -}} {{- with $bird.source -}} {{- range $src := . -}} {{- if or (not $src.label) (not $src.url) }}{{ errorf "%s" (i18n "error_bird_source_incomplete" (dict "ID" (printf "%q" $id))) }}{{ end -}} {{- end -}} {{- else -}} {{- errorf "%s" (i18n "error_bird_source_missing" (dict "ID" (printf "%q" $id))) -}} {{- end -}} {{- $ids = $ids | append $id -}} {{- end -}} {{- $seriesMatched := false -}} {{- range $photo := hugo.Data.photo.photo -}} {{- with $photo.species -}} {{- if not (in $ids .) }}{{ errorf "%s" (i18n "error_photo_species_unknown" (dict "Src" (printf "%q" $photo.src) "Species" (printf "%q" .))) }}{{ end -}} {{- end -}} {{- if and $birdSeries (eq ($photo.series | default "") $birdSeries) -}} {{- $seriesMatched = true -}} {{- end -}} {{- end -}} {{- if and $birdSeries (not $seriesMatched) -}} {{- errorf "%s" (i18n "error_bird_series_unmatched" (dict "Series" (printf "%q" $birdSeries))) -}} {{- end -}} {{- return (dict "series" $birdSeries "levels" $levels "labels" $labels) -}}

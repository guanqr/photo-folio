{{- /* 照片结构化数据（ImageGallery + Photograph 数组，构建期生成零 JS）： 上下文 = dict "photos"（照片对象切片，至少含 src）"title"（页面标题）"url"（页面 URL）。 Photograph 只输出核心元信息文本（名称/描述/拍摄日期/拍摄地点/关键词/作者）， **不含图片 URL/EXIF 明细**：图片 URL 故意不收录（OSS 额度保护）；EXIF 明细 对全页 JSON-LD 体积贡献最大而搜索价值有限，改由灯箱展示。最多取 params.jsonldPhotoLimit 条（默认 24），numberOfItems 仍为真实总数。 */ -}} {{- if .photos -}} {{- $author := site.Params.author -}} {{- $person := dict "@type" "Person" "name" $author.name -}} {{- with site.GetPage "/about" }}{{ $person = merge $person (dict "url" (.Permalink | absURL)) }}{{ end -}} {{- $allPhotos := .photos -}} {{- $limit := 24 -}} {{- /* default 会把配置的 0 当作空值回退 24，0 = 不限制的语义走 isset 判定 */ -}} {{- if isset site.Params "jsonldphotolimit" }}{{ $limit = site.Params.jsonldPhotoLimit | int }}{{ end -}} {{- $photos := $allPhotos -}} {{- if gt $limit 0 }}{{ $photos = first $limit $allPhotos }}{{ end -}} {{- $imgs := slice -}} {{- range $photos -}} {{- $keywords := slice -}} {{- with .category }}{{ $keywords = $keywords | append . }}{{ end -}} {{- with .series }}{{ $keywords = $keywords | append . }}{{ end -}} {{- with .location }}{{ $keywords = $keywords | append . }}{{ end -}} {{- $img := dict "@type" "Photograph" "name" (.alt | default (path.Base .src)) "creator" $person -}} {{- with .description }}{{ $img = merge $img (dict "description" .) }}{{ end -}} {{- with .time }}{{ $img = merge $img (dict "dateCreated" (. | time.Format "2006-01-02")) }}{{ end -}} {{- $loc := or .place .location -}} {{- with $loc }}{{ $img = merge $img (dict "contentLocation" .) }}{{ end -}} {{- if $keywords }}{{ $img = merge $img (dict "keywords" $keywords) }}{{ end -}} {{- $imgs = $imgs | append $img -}} {{- end -}} {{- end -}}
